cmake_minimum_required(VERSION 3.24)
project(SVRPhysicalDoorsRelease VERSION 0.1.41 LANGUAGES CXX C RC)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(COMMONLIB_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/third-party/CommonLibSSE-NG")
set(FMT_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/third-party/fmt")
set(SPDLOG_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/third-party/spdlog")
set(DEPS "" CACHE PATH "vcpkg installed x64-windows prefix")
list(PREPEND CMAKE_PREFIX_PATH "${DEPS}")
find_package(directxtk CONFIG REQUIRED)
find_package(xbyak CONFIG REQUIRED)
add_library(PCVRFmt STATIC "${FMT_SOURCE}/src/format.cc" "${FMT_SOURCE}/src/os.cc")
target_include_directories(PCVRFmt PUBLIC "${FMT_SOURCE}/include")
add_library(PCVRLog STATIC "${SPDLOG_SOURCE}/src/spdlog.cpp" "${SPDLOG_SOURCE}/src/stdout_sinks.cpp" "${SPDLOG_SOURCE}/src/color_sinks.cpp" "${SPDLOG_SOURCE}/src/file_sinks.cpp" "${SPDLOG_SOURCE}/src/async.cpp" "${SPDLOG_SOURCE}/src/cfg.cpp")
target_include_directories(PCVRLog PUBLIC "${SPDLOG_SOURCE}/include")
target_compile_definitions(PCVRLog PUBLIC SPDLOG_COMPILED_LIB SPDLOG_FMT_EXTERNAL)
target_link_libraries(PCVRLog PUBLIC PCVRFmt)
include("${COMMONLIB_SOURCE}/cmake/sourcelist.cmake")
list(TRANSFORM SOURCES PREPEND "${COMMONLIB_SOURCE}/")
add_library(CommonLibVR STATIC ${SOURCES} "${CMAKE_CURRENT_SOURCE_DIR}/third-party/hde64/src/hde/hde64.c")
set_source_files_properties("${CMAKE_CURRENT_SOURCE_DIR}/third-party/hde64/src/hde/hde64.c" PROPERTIES SKIP_PRECOMPILE_HEADERS ON)
target_include_directories(CommonLibVR PUBLIC "${COMMONLIB_SOURCE}/include" "${COMMONLIB_SOURCE}/extern/openvr/headers" "${DEPS}/include" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third-party/hde64/src/hde")
target_compile_definitions(CommonLibVR PUBLIC ENABLE_SKYRIM_VR=1 SKSE_SUPPORT_XBYAK=1 SKSE_SUPPORT_PATCH_SAFETY=1 WINVER=0x0601 _WIN32_WINNT=0x0601 PRIVATE COMMONLIB_VERSION="10.1.0")
target_link_libraries(CommonLibVR PUBLIC PCVRLog Microsoft::DirectXTK xbyak::xbyak advapi32 bcrypt d3d11 d3dcompiler dbghelp dxgi ole32 version "${COMMONLIB_SOURCE}/extern/openvr/lib/win64/openvr_api.lib")
target_precompile_headers(CommonLibVR PRIVATE "${COMMONLIB_SOURCE}/include/SKSE/Impl/PCH.h")

foreach(target SVRPhysicalDoors)
  if(target STREQUAL "SVRPhysicalDoors")
    set(plugin_source physical_doors.cpp)
  else()
    set(plugin_source physical_containers.cpp)
  endif()
  add_library(${target} SHARED "${target}/${plugin_source}")
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${DEPS}/include/directxtk")
  target_link_libraries(${target} PRIVATE CommonLibVR user32 shell32)
  target_precompile_headers(${target} PRIVATE "${COMMONLIB_SOURCE}/include/SKSE/Impl/PCH.h")
endforeach()
foreach(target PCVRFmt PCVRLog CommonLibVR SVRPhysicalDoors)
  target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/permissive-> $<$<COMPILE_LANGUAGE:CXX>:/Zc:preprocessor> $<$<COMPILE_LANGUAGE:CXX>:/Zc:__cplusplus> $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
endforeach()
